Cú pháp @keyframes
/* Cách 1: from → to */ @keyframes fadeIn { from { opacity: 0; transform: translateY(20px); } to { opacity: 1; transform: translateY(0); } } /* Cách 2: dùng phần trăm */ @keyframes bounce { 0% { transform: translateY(0); } 30% { transform: translateY(-28px); } 60% { transform: translateY(-14px); } 100% { transform: translateY(0); } } /* Gán vào element */ .hop { animation: bounce 1s ease-in-out infinite; }

Cấu trúc cơ bản

Các loại selector
p { } /* tag — chọn tất cả thẻ p */ .myClass { } /* class */ #myId { } /* id — chỉ 1 phần tử */ * { } /* tất cả phần tử */ div p { } /* p bên trong div (bất kỳ cấp) */ div > p { } /* p là con trực tiếp của div */ div + p { } /* p ngay sau div */ div ~ p { } /* tất cả p sau div */ div, p { } /* cả div lẫn p */ a:hover { } /* pseudo-class */ p::before { } /* pseudo-element */ [type="text"] { } /* attribute selector */
